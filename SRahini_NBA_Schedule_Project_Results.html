# Ensure gamedate is datetime
schedule['gamedate'] = pd.to_datetime(schedule['gamedate'])

# Function to count 4-in-6 stretches for a team/season
def count_4_in_6_for_team(df):
    df = df.sort_values('gamedate').reset_index(drop=True)
    dates = df['gamedate']
    count = 0
    for i in range(len(dates)):
        window_start = dates[i] - pd.Timedelta(days=5)
        games_in_window = ((dates >= window_start) & (dates <= dates[i])).sum()
        if games_in_window >= 4:
            count += 1
    return count

# Group by team and season, count stretches and adjust to per-82 games
results = []
for (team, season), group in schedule.groupby(['team', 'season']):
    n_games = len(group)
    stretches = count_4_in_6_for_team(group)
    # Adjust to per-82 games
    stretches_per_82 = stretches * 82 / n_games if n_games > 0 else 0
    results.append({'team': team, 'season': season, 'stretches_per_82': stretches_per_82})

results_df = pd.DataFrame(results)

# Average across all team-seasons
average_4_in_6 = results_df['stretches_per_82'].mean()
print(f"Average number of 4-in-6 stretches per team-season (per 82 games): {average_4_in_6:.1f}")

# Outlier test (Grubbs' test for max and min)
def grubbs_test(values, extreme):
    N = len(values)
    mean = values.mean()
    std = values.std()
    G = abs(extreme - mean) / std
    # Critical value for alpha=0.05
    t = stats.t.ppf(1-0.05/(2*N), N-2)
    G_crit = ((N-1)/np.sqrt(N)) * np.sqrt(t**2/(N-2+t**2))
    return G, G_crit

G_max, G_crit_max = grubbs_test(values, most_stretches_value)
G_min, G_crit_min = grubbs_test(values, fewest_stretches_value)

print(f"Grubbs' test statistic for min: {G_min:.2f} (critical: {G_crit_min:.2f})")

def haversine(lat1, lon1, lat2, lon2):
    R = 3959  # miles
    lat1, lon1, lat2, lon2 = map(np.radians, [lat1, lon1, lat2, lon2])
    dlat = lat2 - lat1
    dlon = lon2 - lon1
    a = np.sin(dlat/2)**2 + np.cos(lat1)*np.cos(lat2)*np.sin(dlon/2)**2
    return 2 * R * np.arcsin(np.sqrt(a))

loc_map = locations.set_index('team')[['latitude', 'longitude']].to_dict('index')
travel_stats = []
for (team, season), group in schedule.groupby(['team', 'season']):

    # If home == 1, use team, else use opponent

    coords = [loc_map.get(t, {'latitude': None, 'longitude': None}) for t in cities]

import plotly.graph_objects as go

def plot_team_schedule_lines(schedule_df, locations_df, team, season):
    # Filter for team and season
    df = schedule_df[(schedule_df['team'] == team) & (schedule_df['season'] == season)].copy()
    df = df.sort_values('gamedate').reset_index(drop=True)
    df['gamedate'] = pd.to_datetime(df['gamedate'])
    
    # Get city coordinates for each game
    loc_map = locations_df.set_index('team')[['latitude', 'longitude']].to_dict('index')
    cities = np.where(df['home'] == 1, team, df['opponent'])
    coords = [loc_map.get(t, {'latitude': None, 'longitude': None}) for t in cities]
    df['latitude'] = [c['latitude'] for c in coords]
    df['longitude'] = [c['longitude'] for c in coords]
    
    # Calculate travel distance between games
    def haversine(lat1, lon1, lat2, lon2):
        R = 3959  # miles
        lat1, lon1, lat2, lon2 = map(np.radians, [lat1, lon1, lat2, lon2])
        dlat = lat2 - lat1
        dlon = lon2 - lon1
        a = np.sin(dlat/2)**2 + np.cos(lat1)*np.cos(lat2)*np.sin(dlon/2)**2
        return 2 * R * np.arcsin(np.sqrt(a))
    travel = [0]
    for i in range(1, len(df)):
        if None not in (df.loc[i-1, 'latitude'], df.loc[i-1, 'longitude'], df.loc[i, 'latitude'], df.loc[i, 'longitude']):
            dist = haversine(df.loc[i-1, 'latitude'], df.loc[i-1, 'longitude'], df.loc[i, 'latitude'], df.loc[i, 'longitude'])
        else:
            dist = 0
        travel.append(dist)
    df['travel_miles'] = travel

    # Calculate games in last 6 days (density)
    density = []
    for i in range(len(df)):
        window_start = df.loc[i, 'gamedate'] - pd.Timedelta(days=5)
        games_in_window = ((df['gamedate'] >= window_start) & (df['gamedate'] <= df.loc[i, 'gamedate'])).sum()
        density.append(games_in_window)
    df['games_in_6days'] = density

    # Plot with filled area under lines and dual y-axes
    fig = go.Figure()

    # Game Density (left y-axis)
    fig.add_trace(go.Scatter(
        x=df['gamedate'], y=df['games_in_6days'],
